<script>
    // 模拟数据库：这里存放你想要“在库”的号码
    const database = ["19183521762", "13800138000", "15900000000"];

    function checkPhone() {
        const input = document.getElementById('phoneInput').value.trim();
        const resultBox = document.getElementById('resultBox');

        if (input === "") {
            alert("请输入号码！");
            return;
        }

        resultBox.style.display = "block";

        // 判断逻辑
        if (database.includes(input)) {
            resultBox.className = "result ok";
            resultBox.innerHTML = `号码 ${input} <br> <b>在库 ✅</b>`;
        } else {
            resultBox.className = "result no";
            resultBox.innerHTML = `号码 ${input} <br> <b>未找到 ❌</b>`;
        }
    }
</script>
